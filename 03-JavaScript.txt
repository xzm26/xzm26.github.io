function calculate({current,baseline,days,value,fare,rent}){const hours=(current-baseline)*2*days/60;const time=hours*value;const travel=fare*days;return{hours,time,travel,rent:rent*12,balance:rent*12-time-travel,breakEven:(time+travel)/12};}
const ids=['current','baseline','days','value','fare','rent'];const units=['分钟','分钟','天','元 / 小时'];const money=n=>`${Math.round(n).toLocaleString('zh-CN')} 元`;
function updateCalc(){const inputs=Object.fromEntries(ids.map(id=>{const el=document.getElementById(id);return[id,Math.min(Number(el.max),Math.max(Number(el.min),Number(el.value)||0))];}));ids.slice(0,4).forEach((id,i)=>document.getElementById(id+'-out').textContent=inputs[id]+' '+units[i]);const r=calculate(inputs);document.getElementById('hours-result').textContent=Number(r.hours.toFixed(1)).toLocaleString('zh-CN');document.getElementById('time-result').textContent=money(r.time);document.getElementById('fare-result').textContent=money(r.travel);document.getElementById('rent-result').textContent=money(r.rent);document.getElementById('balance-result').textContent=money(r.balance);document.getElementById('break-even').textContent=`在这些假设下，每月租金节省达到 ${money(r.breakEven)} 时，两方案的年度总账持平。`;}
ids.forEach(id=>document.getElementById(id).addEventListener('input',updateCalc));document.getElementById('calc-form').addEventListener('reset',()=>setTimeout(updateCalc,0));document.getElementById('calc-form').addEventListener('submit',e=>e.preventDefault());updateCalc();
